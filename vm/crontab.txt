SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
# Times in UTC. Paris CEST (Mar-Oct) = UTC+2. Paris CET (Oct-Mar) = UTC+1.
#
# Jobs are spaced 5 HOURS apart on purpose: the VM authenticates the `claude` CLI with a
# Claude *subscription* token (CLAUDE_CODE_OAUTH_TOKEN), whose usage limit resets on a rolling
# 5-hour window. Bunching the heavy runs (scrape, agent) together would exhaust one window and
# make later runs fail. One heavy job per window → each starts with a fresh allowance.
# All five fire on the same UTC weekday (Mon-Fri), so there is no weekend-boundary skip.

# 04:00 Paris (CEST) = 02:00 UTC — Early inbox scan: catch overnight replies, alert
0 2 * * 1-5  __REPO_DIR__/vm/run_followup.sh >> __REPO_DIR__/logs/cron.log 2>&1

# 09:00 Paris (CEST) = 07:00 UTC — Main outreach agent (SEND up to 7 cold + 3 warm, warm-up ramped)
0 7 * * 1-5  __REPO_DIR__/vm/run_agent.sh >> __REPO_DIR__/logs/cron.log 2>&1

# 14:00 Paris (CEST) = 12:00 UTC — Speculative outreach batch (proactive pitches)
0 12 * * 1-5  __REPO_DIR__/vm/run_speculative.sh >> __REPO_DIR__/logs/cron.log 2>&1

# 19:00 Paris (CEST) = 17:00 UTC — Enrich remaining generic contact@ emails (for tomorrow)
0 17 * * 1-5  __REPO_DIR__/vm/run_find_contacts.sh >> __REPO_DIR__/logs/cron.log 2>&1

# 00:00 Paris (CEST, next day) = 22:00 UTC — Scrape new jobs, refill Pending leads.
# MONDAY + THURSDAY ONLY (was nightly). The 2026-09 audit found 1,380 Pending leads against a
# ceiling of 7 cold sends/day — roughly ten months of work already queued. Scraping five boards
# nightly grew a backlog that can never be drained, while consuming a 5h Claude window that
# conversion steps (enrichment, follow-up) need more. Twice weekly still refills faster than the
# sender can spend. Restore nightly (`* * 1-5`) only if Pending ever drops below ~200.
0 22 * * 1,4  __REPO_DIR__/vm/run_scrape.sh >> __REPO_DIR__/logs/cron.log 2>&1

# 08:30 Paris (CEST) = 06:30 UTC — Stalled warm-lead alert. A human reply is the rarest thing
# this system produces (~1 per 50 delivered cold emails); the audit found them going quiet for
# weeks because the stalled flag was only ever displayed, never pushed. Pure Python, no Claude,
# self-dedupes on a 7-day cooldown per lead, and only sends when something is actually due.
30 6 * * 1-5  cd __REPO_DIR__ && __REPO_DIR__/venv/bin/python stalled_alert.py --send >> __REPO_DIR__/logs/stalled.log 2>&1

# 08:00 Paris (CEST) = 06:00 UTC — International opportunity SCOUT: email Zineb new remote/intl
# offers that fit her profile. Pure Python (no Claude, no git push), so it's cheap and can run
# outside the 5h-spaced Claude windows. Only emails when there are new matches (dedup + --min).
0 6 * * 1-5  cd __REPO_DIR__ && __REPO_DIR__/venv/bin/python opportunities.py --send >> __REPO_DIR__/logs/opportunities.log 2>&1
